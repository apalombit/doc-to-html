Worker loop
while True:
    job = queue.pop(timeout=30)
    if job is None:
        continue
    try:
        run(job.command)
    except BuildError:
        job.retries += 1
        if job.retries <= 2:
            queue.push(job)
        else:
            alert(job.owner)
